<script>
(function(scope){
let p={},received=Date.now();scope.tpsPct=0;scope.lapText='00:00.000';scope.remainingText='00:00.000';
function fmt(seconds){const m=Math.round(Math.abs(seconds)*1000);return (seconds<0?'-':'')+String(Math.floor(m/60000)).padStart(2,'0')+':'+String(Math.floor(m/1000)%60).padStart(2,'0')+'.'+String(m%1000).padStart(3,'0');}
function draw(elapsed){
const points=p.speedHistory||[],xmax=Math.max(250,Math.ceil(elapsed/50)*50),ymax=Math.max(40,Math.ceil(points.reduce((v,q)=>Math.max(v,q.v),0)/10)*10),x=t=>42+t/xmax*788,y=v=>125-v/ymax*110;
scope.grid=[0,1,2,3,4].map(i=>({y:y(ymax*i/4),label:ymax*i/4}));scope.ticks=[0,1,2,3,4,5].map(i=>{const s=Math.round(i*xmax/5);return {x:x(s),label:Math.floor(s/60)+':'+String(s%60).padStart(2,'0')};});
function path(key){let previous=null;return points.map(q=>{if(q[key]===null||!Number.isFinite(q[key])){previous=null;return '';}const head=!previous||q.t-previous.t>2?'M':'L';previous=q;return head+x(q.t).toFixed(2)+','+y(q[key]).toFixed(2);}).join(' ');}
scope.speedPath=path('v');scope.avgPath=path('avg');
}
function tick(){const since=(Date.now()-received)/1000,elapsed=p.isRunning?(Number(p.elapsedSec)||0)+since:0;scope.lapText=p.isRunning?fmt(elapsed):'00:00.000';const remaining=(Number(p.effectiveTarget)||200)-elapsed;scope.remainingText=p.isRunning?fmt(remaining):'00:00.000';scope.remainingColor=!p.isRunning?'#8091a5':remaining<=0?'#ff647c':remaining<60?'#ffd36b':'#67e7ab';scope.gpsStatus=p.gpsAge==null?'Menunggu GPS':Number(p.gpsAge)+since>2?'GPS terlambat / terputus':'GPS LIVE';}
scope.$watch('msg',function(m){if(!m||!m.payload)return;p=m.payload;received=Date.now();const v=Number(p.tps);scope.tpsPct=Number.isFinite(v)?Math.max(0,Math.min(100,v)):0;tick();draw(Number(p.elapsedSec)||0);});
const timer=setInterval(function(){scope.$evalAsync(tick);},100);scope.$on('$destroy',function(){clearInterval(timer);});draw(0);
})(scope);
</script>
